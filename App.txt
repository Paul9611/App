                'In Production': '<span class="px-2.5 py-1 bg-indigo-100 text-indigo-800 rounded-full text-[10px] font-bold">In Production</span>',

        // Populate and Preview Production Run Modal
        function populateProductionDropdown() {
            const select = document.getElementById('run-product-id');
            select.innerHTML = state.products.map(p => `<option value="${p.id}">${p.name} (Current Stock: ${p.stock})</option>`).join('');
            updateProductionPreview();

        function updateProductionPreview() {
            const prodId = document.getElementById('run-product-id').value;
            const qty = parseInt(document.getElementById('run-qty').value) || 0;
            const preview = document.getElementById('production-deduction-preview');
            const submitBtn = document.getElementById('btn-submit-run');

            const prod = state.products.find(p => p.id === prodId);
            if (!prod || !prod.recipe || prod.recipe.length === 0) {
                preview.innerHTML = '<span class="text-amber-700 italic">No recipe defined for this product. Production will add finished stock without raw material deduction.</span>';
                submitBtn.disabled = false;
                return;
            }

            let html = '<ul class="list-disc pl-4 space-y-1">';
            let hasSufficientStock = true;
            let summaryList = [];

            prod.recipe.forEach(r => {
                const mat = state.materials.find(m => m.id === r.materialId);
                const totalGramsNeeded = r.grams * qty;
                const totalKgNeeded = totalGramsNeeded / 1000;

                if (mat) {
                    const isEnough = mat.weightKg >= totalKgNeeded;
                    if (!isEnough) hasSufficientStock = false;

                    html += `
                        <li class="${isEnough ? 'text-amber-900' : 'text-red-600 font-bold'}">
                            ${mat.name}: Requires <strong>${totalKgNeeded.toFixed(2)} kg</strong> (${totalGramsNeeded}g) &mdash; 
                            Available: ${mat.weightKg.toFixed(2)} kg ${isEnough ? '✓' : '⚠️ INSUFFICIENT RAW PLASTIC'}
                        </li>
                    `;
                    summaryList.push(`${totalKgNeeded.toFixed(2)}kg ${mat.type}`);
                }
            });
            html += '</ul>';

            if (!hasSufficientStock) {
                html += '<p class="mt-2 text-red-600 font-bold">Cannot execute batch: Insufficient raw plastic material stock.</p>';
                submitBtn.disabled = true;
                submitBtn.classList.add('opacity-50', 'cursor-not-allowed');
            } else {
                submitBtn.disabled = false;
                submitBtn.classList.remove('opacity-50', 'cursor-not-allowed');
            }

            preview.innerHTML = html;
        }

        // Execute Production Batch Run & Automatically Deduct Raw Plastics
        function executeProductionRun(e) {
            e.preventDefault();
            const prodId = document.getElementById('run-product-id').value;
            const qty = parseInt(document.getElementById('run-qty').value) || 0;
            const notes = document.getElementById('run-notes').value;

            const prod = state.products.find(p => p.id === prodId);
            if (!prod) return;

            let summaryDeductions = [];

            // 1. Deduct Materials
            if (prod.recipe) {
                prod.recipe.forEach(r => {
                    const mat = state.materials.find(m => m.id === r.materialId);
                    if (mat) {
                        const totalKgNeeded = (r.grams * qty) / 1000;
                        mat.weightKg = Math.max(0, mat.weightKg - totalKgNeeded);
                        summaryDeductions.push(`${totalKgNeeded.toFixed(2)}kg ${mat.name}`);
                    }
                });
            }

            // 2. Add Stock to Finished Product
            prod.stock += qty;

            // 3. Log Production Run
            const now = new Date();
            const dateStr = `${now.toISOString().split('T')[0]} ${now.toTimeString().split(' ')[0].substring(0, 5)}`;
            
            state.productionRuns.unshift({
                id: `pr-${Date.now()}`,
                date: dateStr,
                productId: prod.id,
                productName: prod.name,
                qtyProduced: qty,
                materialsUsedSummary: summaryDeductions.join(', ') || 'None',
                notes: notes
            });

            saveState();
            closeModal('modal-production');
            document.getElementById('form-production').reset();
        }

        // Order Line Items Helper
        function addOrderLineRow(productId = '', qty = 1) {
            const container = document.getElementById('order-line-items-container');
            const rowId = `order-line-${Date.now()}-${Math.random().toString(36).substr(2, 4)}`;

            const options = state.products.map(p => `
                <option value="${p.id}" ${p.id === productId ? 'selected' : ''}>${p.name} (£${p.price.toFixed(2)} / Stock: ${p.stock})</option>
            `).join('');

            const html = `
                <div id="${rowId}" class="flex items-center gap-2">
                    <select onchange="calculateOrderTotal()" class="order-prod-select flex-1 border border-slate-300 rounded-lg p-2 text-xs">
                        ${options}
                    </select>
                    <input type="number" value="${qty}" min="1" oninput="calculateOrderTotal()" placeholder="Qty" class="order-qty-input w-20 border border-slate-300 rounded-lg p-2 text-xs">
                    <button type="button" onclick="document.getElementById('${rowId}').remove(); calculateOrderTotal();" class="text-red-500 hover:bg-red-50 p-1 rounded">&times;</button>
                </div>
            `;
            container.insertAdjacentHTML('beforeend', html);
            calculateOrderTotal();
        }

        function calculateOrderTotal() {
            let total = 0;
            const rows = document.querySelectorAll('#order-line-items-container > div');
            rows.forEach(r => {
                const select = r.querySelector('.order-prod-select');
                const qtyInput = r.querySelector('.order-qty-input');
                if (select && qtyInput) {
                    const prod = state.products.find(p => p.id === select.value);
                    const qty = parseInt(qtyInput.value) || 0;
                    if (prod) {
                        total += prod.price * qty;
                    }
                }
            });
            document.getElementById('order-total').value = total.toFixed(2);
        }

        function saveOrder(e) {
            e.preventDefault();
            const id = document.getElementById('order-id').value || `ORD-${Math.floor(1000 + Math.random() * 9000)}`;
            const status = document.getElementById('order-status').value;
            
            const items = [];
            const rows = document.querySelectorAll('#order-line-items-container > div');
            rows.forEach(r => {
                const select = r.querySelector('.order-prod-select');
                const qtyInput = r.querySelector('.order-qty-input');
                if (select && qtyInput) {
                    const prod = state.products.find(p => p.id === select.value);
                    items.push({
                        productId: select.value,
                        qty: parseInt(qtyInput.value) || 1,
                        price: prod ? prod.price : 0
                    });
                }
            });

            const existingOrder = state.orders.find(o => o.id === id);

            // Deduct finished goods stock if order status transitions to Completed/Shipped
            if ((status === 'Completed' || status === 'Shipped') && (!existingOrder || existingOrder.status === 'Pending' || existingOrder.status === 'In Production')) {
                items.forEach(item => {
                    const prod = state.products.find(p => p.id === item.productId);
                    if (prod) {
                        prod.stock = Math.max(0, prod.stock - item.qty);
                    }
                });
            }

            const newOrder = {
                id: id,
                customer: document.getElementById('order-customer').value,
                contact: document.getElementById('order-contact').value,
                items: items,
                total: parseFloat(document.getElementById('order-total').value) || 0,
                status: status,
                date: existingOrder ? existingOrder.date : new Date().toISOString().split('T')[0]
            };

            const idx = state.orders.findIndex(o => o.id === id);
            if (idx >= 0) state.orders[idx] = newOrder;
            else state.orders.unshift(newOrder);

            saveState();
            closeModal('modal-order');
            document.getElementById('form-order').reset();
            document.getElementById('order-id').value = '';
            document.getElementById('order-line-items-container').innerHTML = '';
        }

        function editOrder(id) {
            const o = state.orders.find(item => item.id === id);
            if (!o) return;
            document.getElementById('order-id').value = o.id;
            document.getElementById('order-customer').value = o.customer;
            document.getElementById('order-contact').value = o.contact || '';

            container.innerHTML = '';
            o.items.forEach(i => addOrderLineRow(i.productId, i.qty));
            
            openModal('modal-order');
        }

        function deleteOrder(id) {

        }
